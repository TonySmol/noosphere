/** * NOOmium Engine Core * Изолированное ядро приложения с DI, EventBus и плагинной системой * @module noomium-engine */ // ============================================================================ // LOGGER // ============================================================================ class Logger { constructor(debug = false) { this.debug = debug; this._enabled = true; } enable() { this._enabled = true; } disable() { this._enabled = false; } info(msg, data) { if (!this._enabled) return; console.log(`[INFO] ${msg}`, data !== undefined ? data : ''); } warn(msg, data) { if (!this._enabled) return; console.warn(`[WARN] ${msg}`, data !== undefined ? data : ''); } error(msg, data) { if (!this._enabled) return; console.error(`[ERROR] ${msg}`, data !== undefined ? data : ''); } debug_log(msg, data) { if (!this._enabled || !this.debug) return; console.log(`[DEBUG] ${msg}`, data !== undefined ? data : ''); } } // ============================================================================ // ENGINE CORE // ============================================================================ class Engine { /** * @param {Object} options * @param {boolean} options.debug - включить debug-логи * @param {Object} options.config - конфигурация по умолчанию */ constructor(options = {}) { this.debug = options.debug || false; this.config = { ...options.config }; // DI Container this._factories = new Map(); this._instances = new Map(); this._resolving = new Set(); // для детекции циклических зависимостей // EventBus this._listeners = new Map(); this._eventRegistry = new Map(); // зарегистрированные события с описаниями // Module status tracking this._moduleStatus = new Map(); // Plugins this._plugins = []; // Logger this.log = new Logger(this.debug); // Timestamps this._startTime = Date.now(); this._started = false; } // ========================================================================== // DI CONTAINER // ========================================================================== /** * Регистрация сервиса в DI-контейнере * @param {string} name - имя сервиса * @param {Function} factory - фабрика (engine) => service * @param {string[]} deps - список зависимостей (имена других сервисов) */ register(name, factory, deps = []) { if (this._factories.has(name)) { this.log.warn(`Service "${name}" re-registered`); } this._factories.set(name, { factory, deps }); this._setModuleStatus(name, 'pending'); } /** * Получение инстанса сервиса (lazy initialization) * @param {string} name * @returns {any} */ get(name) { // Возвращаем кэшированный инстанс if (this._instances.has(name)) { return this._instances.get(name); } const def = this._factories.get(name); if (!def) { const err = new Error(`Service "${name}" not registered`); this.log.error(err.message); throw err; } // Детекция циклических зависимостей if (this._resolving.has(name)) { const err = new Error(`Circular dependency detected: ${name}`); this.log.error(err.message); this._setModuleStatus(name, 'error', err.message); throw err; } this._resolving.add(name); try { // Резолвим зависимости const resolvedDeps = def.deps.map(depName => this.get(depName)); // Вызываем фабрику const instance = def.factory(this, ...resolvedDeps); this._instances.set(name, instance); this._setModuleStatus(name, 'ready'); return instance; } catch (err) { this._setModuleStatus(name, 'error', err.message); this.log.error(`Failed to initialize service "${name}"`, err); throw err; } finally { this._resolving.delete(name); } } /** * Проверка, зарегистрирован ли сервис * @param {string} name * @returns {boolean} */ has(name) { return this._factories.has(name); } // ========================================================================== // EVENT BUS // ========================================================================== /** * Регистрация типа события (для документации и проверок) * @param {string} eventName * @param {string} description */ registerEvent(eventName, description = '') { this._eventRegistry.set(eventName, { description, registeredAt: Date.now() }); } /** * Подписка на событие * @param {string} event * @param {Function} handler */ on(event, handler) { if (this.debug && !this._eventRegistry.has(event)) { this.log.warn(`Event "${event}" not registered (use registerEvent for documentation)`); } if (!this._listeners.has(event)) { this._listeners.set(event, new Set()); } this._listeners.get(event).add(handler); } /** * Отписка от события * @param {string} event * @param {Function} handler */ off(event, handler) { const set = this._listeners.get(event); if (set) { set.delete(handler); if (set.size === 0) { this._listeners.delete(event); } } } /** * Эмиссия события * @param {string} event * @param {any} payload */ emit(event, payload) { const set = this._listeners.get(event); if (!set) return; for (const handler of set) { try { handler(payload); } catch (err) { this.log.error(`Event handler error for "${event}"`, err); } } } /** * Подписка на событие с автоматической отпиской (one-time) * @param {string} event * @param {Function} handler */ once(event, handler) { const wrapper = (payload) => { this.off(event, wrapper); handler(payload); }; this.on(event, wrapper); } // ========================================================================== // PLUGIN SYSTEM // ========================================================================== /** * Установка плагина * @param {Function} installFn - (engine) => void */ plugin(installFn) { if (typeof installFn !== 'function') { throw new Error('Plugin must be a function'); } this._plugins.push(installFn); try { installFn(this); this.log.debug_log(`Plugin installed`); } catch (err) { this.log.error('Plugin installation failed', err); throw err; } } // ========================================================================== // LIFECYCLE // ========================================================================== /** * Запуск движка (инициализация всех модулей) * @returns {Promise
